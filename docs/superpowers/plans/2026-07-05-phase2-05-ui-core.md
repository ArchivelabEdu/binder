```ts
export async function getRelationCounts(
  db: PostgresJsDatabase,
  nodeIds: number[],
): Promise<Map<number, number>> {
  if (nodeIds.length === 0) return new Map();
  const rows = (await db.execute(sql`
    SELECT r.subject_node_id AS sid, count(*)::int AS c
    FROM relations r JOIN nodes o ON o.id = r.object_node_id AND o.deleted_at IS NULL
    WHERE r.subject_node_id IN ${sql`(${sql.join(nodeIds.map((i) => sql`${i}`), sql`, `)})`}
    GROUP BY r.subject_node_id
  `)) as unknown as Array<{ sid: number; c: number }>;
  return new Map(rows.map((r) => [Number(r.sid), Number(r.c)]));
}
```

```ts
import { sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import type { I18nText } from '../types.js';
import { searchEntities } from './search.js';

export type BrowseNodesOpts = {
  kind?: 'work' | 'component' | 'record' | 'supporting';
  levelTermId?: number;
  departmentTermId?: number;
  q?: string;
  page?: number;
  pageSize?: number;
};

export type BrowseItem = {
  id: number; uuid: string; slug: string; kind: string; title: I18nText;
  accessionNumber: string | null; identifier: string | null;
  levelTermId: number; departmentTermId: number | null; thumbnailUrl: string | null;
  aipCount: number; totalSizeBytes: number; updatedAt: Date;
};

export type BrowseNodesResult = {
  items: BrowseItem[];
  total: number;
  facets: {
    kinds: Array<{ key: string; count: number }>;
    levels: Array<{ termId: number; count: number }>;
    departments: Array<{ termId: number; count: number }>;
  };
};

export async function browseNodes(db: PostgresJsDatabase, opts: BrowseNodesOpts): Promise<BrowseNodesResult> {
  const page = Math.max(1, opts.page ?? 1);
  const pageSize = Math.min(Math.max(1, opts.pageSize ?? 20), 100);

  let searchIds: number[] | null = null;
  if (opts.q?.trim()) {
    const hits = await searchEntities(db, opts.q, { entityType: 'node', limit: 500 });
    searchIds = hits.map((h) => h.entityId);
    if (searchIds.length === 0) {
      return { items: [], total: 0, facets: { kinds: [], levels: [], departments: [] } };
    }
  }

  const conds = [sql`n.deleted_at IS NULL`];
  if (opts.kind) conds.push(sql`n.kind = ${opts.kind}`);
  else conds.push(sql`n.kind <> 'component'`);
  if (opts.levelTermId) conds.push(sql`n.level_term_id = ${opts.levelTermId}`);
  if (opts.departmentTermId) conds.push(sql`n.department_term_id = ${opts.departmentTermId}`);
  if (searchIds) conds.push(sql`n.id IN ${sql`(${sql.join(searchIds.map((i) => sql`${i}`), sql`, `)})`}`);
  const where = sql.join(conds, sql` AND `);

  const items = (await db.execute(sql`
    SELECT n.id, n.uuid, n.slug, n.kind, n.title, n.accession_number, n.identifier,
           n.level_term_id, n.department_term_id, n.thumbnail_url,
           n.aip_count, n.total_size_bytes, n.updated_at
    FROM nodes n WHERE ${where}
    ORDER BY n.updated_at DESC
    LIMIT ${pageSize} OFFSET ${(page - 1) * pageSize}
  `)) as unknown as Array<Record<string, unknown>>;

  const [{ total }] = (await db.execute(sql`
    SELECT count(*)::int AS total FROM nodes n WHERE ${where}
  `)) as unknown as Array<{ total: number }>;

  const kinds = (await db.execute(sql`
    SELECT n.kind AS key, count(*)::int AS count FROM nodes n WHERE ${where} GROUP BY n.kind
  `)) as unknown as Array<{ key: string; count: number }>;
  const levels = (await db.execute(sql`
    SELECT n.level_term_id AS term_id, count(*)::int AS count FROM nodes n WHERE ${where} GROUP BY n.level_term_id
  `)) as unknown as Array<{ term_id: number; count: number }>;
  const departments = (await db.execute(sql`
    SELECT n.department_term_id AS term_id, count(*)::int AS count
    FROM nodes n WHERE ${where} AND n.department_term_id IS NOT NULL
    GROUP BY n.department_term_id
  `)) as unknown as Array<{ term_id: number; count: number }>;

  let mapped: BrowseItem[] = items.map((r) => ({
    id: Number(r.id), uuid: String(r.uuid), slug: String(r.slug), kind: String(r.kind),
    title: r.title as I18nText,
    accessionNumber: r.accession_number === null ? null : String(r.accession_number),
    identifier: r.identifier === null ? null : String(r.identifier),
    levelTermId: Number(r.level_term_id),
    departmentTermId: r.department_term_id === null ? null : Number(r.department_term_id),
    thumbnailUrl: r.thumbnail_url === null ? null : String(r.thumbnail_url),
    aipCount: Number(r.aip_count), totalSizeBytes: Number(r.total_size_bytes),
    updatedAt: new Date(r.updated_at as string),
  }));
  if (searchIds) {
    const order = new Map(searchIds.map((id, i) => [id, i]));
    mapped = mapped.sort((a, b) => (order.get(a.id) ?? 1e9) - (order.get(b.id) ?? 1e9));
  }

  return {
    items: mapped,
    total: Number(total),
    facets: {
      kinds: kinds.map((k) => ({ key: k.key, count: Number(k.count) })),
      levels: levels.map((l) => ({ termId: Number(l.term_id), count: Number(l.count) })),
      departments: departments.map((d) => ({ termId: Number(d.term_id), count: Number(d.count) })),
    },
  };
}
```

```ts
import { inArray } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import { terms } from '../schema/index.js';
import type { I18nText } from '../types.js';

export async function getTermNames(db: PostgresJsDatabase, termIds: number[]): Promise<Map<number, I18nText>> {
  if (termIds.length === 0) return new Map();
  const rows = await db.select({ id: terms.id, name: terms.name }).from(terms).where(inArray(terms.id, termIds));
  return new Map(rows.map((r) => [r.id, r.name]));
}
```

```tsx
export function Pagination({ page, total, pageSize, baseQuery, prevLabel, nextLabel }: {
  page: number; total: number; pageSize: number; baseQuery: Record<string, string>;
  prevLabel: string; nextLabel: string;
}) {
  const last = Math.max(1, Math.ceil(total / pageSize));

  };

      {page > 1 && <Link className="rounded border px-2 py-1" href={href(page - 1)}>{prevLabel}</Link>}
      <span className="text-neutral-500">{page} / {last}</span>
      {page < last && <Link className="rounded border px-2 py-1" href={href(page + 1)}>{nextLabel}</Link>}
    </div>
  );
}
```

```ts
import { and, eq, isNull } from 'drizzle-orm';
import { sql } from 'drizzle-orm';
import type { PostgresJsDatabase } from 'drizzle-orm/postgres-js';
import { agents, aips, nodes, notes, nodeRights, rights, rightsActs, relations, terms } from '../schema/index.js';
import type { I18nText } from '../types.js';
import { getCreators } from './events.js';
import { getSubtree } from './nodes.js';
import { latestFixityByAip } from './fixity.js';
import { getTermNames } from './terms.js';

export type NodeDetail = {
  node: {
    id: number; uuid: string; slug: string; kind: string; title: I18nText;
    identifier: string | null; accessionNumber: string | null; componentNumber: string | null;
    descriptive: Record<string, unknown> | null;
    sourceAttributes: Array<{ label: string; value: string }> | null;
    dateCreated: string | null; thumbnailUrl: string | null;
    aipCount: number; totalSizeBytes: number; fileCount: number;
    levelTermId: number; departmentTermId: number | null;
  };
  levelName: I18nText | null;
  departmentName: I18nText | null;
  creators: Array<{ agentId: number; name: I18nText }>;
  dateDisplay: I18nText | null;
  notes: Array<{ typeName: I18nText; content: I18nText }>;
  rights: Array<{ basisName: I18nText; holderName: I18nText | null; acts: Array<{ actName: I18nText; restriction: string }> }>;
  relations: Array<{ direction: 'out' | 'in'; typeName: I18nText; otherId: number; otherSlug: string; otherTitle: I18nText }>;
  aips: Array<{ id: number; uuid: string; filename: string | null; sizeBytes: number; typeName: I18nText | null; createdAt: Date; fixity: { success: boolean; completedAt: Date | null } | null }>;
};

export async function getNodeDetail(db: PostgresJsDatabase, slug: string): Promise<NodeDetail | null> {
  const [n] = await db.select().from(nodes).where(and(eq(nodes.slug, slug), isNull(nodes.deletedAt)));
  if (!n) return null;

  const termIds = [n.levelTermId, n.departmentTermId].filter((v): v is number => v != null);

  // 이벤트 date_display (제작 이벤트)
  const [ev] = (await db.execute(sql`
    SELECT e.date_display FROM events e
    JOIN terms t ON t.id = e.type_term_id AND t.machine_name = 'creation'
    WHERE e.node_id = ${n.id} LIMIT 1
  `)) as unknown as Array<{ date_display: I18nText | null }>;

  const noteRows = (await db.execute(sql`
    SELECT tt.name AS type_name, no.content
    FROM notes no JOIN terms tt ON tt.id = no.type_term_id
    WHERE no.node_id = ${n.id} ORDER BY no.id
  `)) as unknown as Array<{ type_name: I18nText; content: I18nText }>;

  const rightRows = (await db.execute(sql`
    SELECT r.id, bt.name AS basis_name, ag.name AS holder_name
    FROM node_rights nr
    JOIN rights r ON r.id = nr.rights_id
    JOIN terms bt ON bt.id = r.basis_term_id
    LEFT JOIN agents ag ON ag.id = r.rights_holder_agent_id AND ag.deleted_at IS NULL
    WHERE nr.node_id = ${n.id}
  `)) as unknown as Array<{ id: number; basis_name: I18nText; holder_name: I18nText | null }>;
  const rightsOut: NodeDetail['rights'] = [];
  for (const r of rightRows) {
    const acts = (await db.execute(sql`
      SELECT at.name AS act_name, ra.restriction
      FROM rights_acts ra JOIN terms at ON at.id = ra.act_term_id
      WHERE ra.rights_id = ${r.id} ORDER BY ra.id
    `)) as unknown as Array<{ act_name: I18nText; restriction: string }>;
    rightsOut.push({
      basisName: r.basis_name, holderName: r.holder_name,
      acts: acts.map((a) => ({ actName: a.act_name, restriction: String(a.restriction) })),
    });
  }

  const relRows = (await db.execute(sql`
    SELECT 'out' AS direction, rt.name AS type_name, o.id AS other_id, o.slug AS other_slug, o.title AS other_title
    FROM relations r
    JOIN terms rt ON rt.id = r.type_term_id
    JOIN nodes o ON o.id = r.object_node_id AND o.deleted_at IS NULL
    WHERE r.subject_node_id = ${n.id}
    UNION ALL
    SELECT 'in', rt.name, s.id, s.slug, s.title
    FROM relations r
    JOIN terms rt ON rt.id = r.type_term_id
    JOIN nodes s ON s.id = r.subject_node_id AND s.deleted_at IS NULL
    WHERE r.object_node_id = ${n.id}
  `)) as unknown as Array<{ direction: 'out' | 'in'; type_name: I18nText; other_id: number; other_slug: string; other_title: I18nText }>;

  const aipRows = (await db.execute(sql`
    SELECT a.id, a.uuid, a.filename, a.size_bytes, a.created_at, t.name AS type_name
    FROM aips a LEFT JOIN terms t ON t.id = a.type_term_id
    WHERE (a.node_id = ${n.id} OR a.root_node_id = ${n.id}) AND a.deleted_at IS NULL
    ORDER BY a.created_at DESC
  `)) as unknown as Array<Record<string, unknown>>;
  const fixity = await latestFixityByAip(db, aipRows.map((a) => Number(a.id)));

  const names = await getTermNames(db, termIds);
  const creators = await getCreators(db, n.id);

  return {
    node: {
      id: n.id, uuid: n.uuid, slug: n.slug, kind: n.kind, title: n.title,
      identifier: n.identifier, accessionNumber: n.accessionNumber, componentNumber: n.componentNumber,
      descriptive: n.descriptive ?? null, sourceAttributes: n.sourceAttributes ?? null,
      dateCreated: n.dateCreated, thumbnailUrl: n.thumbnailUrl,
      aipCount: n.aipCount, totalSizeBytes: n.totalSizeBytes, fileCount: n.fileCount,
      levelTermId: n.levelTermId, departmentTermId: n.departmentTermId,
    },
    levelName: names.get(n.levelTermId) ?? null,
    departmentName: n.departmentTermId ? names.get(n.departmentTermId) ?? null : null,
    creators,
    dateDisplay: ev?.date_display ?? null,
    notes: noteRows.map((r) => ({ typeName: r.type_name, content: r.content })),
    rights: rightsOut,
    relations: relRows.map((r) => ({
      direction: r.direction, typeName: r.type_name,
      otherId: Number(r.other_id), otherSlug: String(r.other_slug), otherTitle: r.other_title,
    })),
    aips: aipRows.map((a) => ({
      id: Number(a.id), uuid: String(a.uuid),
      filename: a.filename === null ? null : String(a.filename),
      sizeBytes: Number(a.size_bytes), typeName: (a.type_name ?? null) as I18nText | null,
      createdAt: new Date(a.created_at as string),
      fixity: fixity.get(Number(a.id)) ?? null,
    })),
  };
}

export type ContextGraph = {
  nodes: Array<{ id: string; kind: 'record' | 'aip' | 'supporting'; label: I18nText; slug?: string;
    meta?: { level?: I18nText; sizeBytes?: number; componentNumber?: string }; x: number; y: number }>;
  edges: Array<{ id: string; source: string; target: string; kind: 'tree' | 'aip' | 'relation'; label?: I18nText }>;
};

const COL_W = 280;
const ROW_H = 96;

export async function getContextGraph(db: PostgresJsDatabase, rootNodeId: number): Promise<ContextGraph> {
  const tree = await getSubtree(db, rootNodeId);
  if (tree.length === 0) return { nodes: [], edges: [] };

  const levelNames = await getTermNames(db, [...new Set(tree.map((t) => t.levelTermId))]);
  const treeIds = tree.map((t) => t.id);

  const compRows = (await db.execute(sql`
    SELECT id, component_number FROM nodes
    WHERE id IN ${sql`(${sql.join(treeIds.map((i) => sql`${i}`), sql`, `)})`}
  `)) as unknown as Array<{ id: number; component_number: string | null }>;
  const compByNode = new Map(compRows.map((r) => [Number(r.id), r.component_number]));

  const gNodes: ContextGraph['nodes'] = [];
  const gEdges: ContextGraph['edges'] = [];
  const usedY = new Map<number, number>(); // x열별 다음 y 슬롯

  tree.forEach((t, i) => {
    gNodes.push({
      id: `n${t.id}`, kind: 'record', label: t.title, slug: t.slug,
      meta: {
        level: levelNames.get(t.levelTermId),
        componentNumber: compByNode.get(t.id) ?? undefined,
      },
      x: t.depth * COL_W, y: i * ROW_H,
    });
    if (t.parentId !== null && tree.some((p) => p.id === t.parentId)) {
      gEdges.push({ id: `t${t.id}`, source: `n${t.parentId}`, target: `n${t.id}`, kind: 'tree' });
    }
  });

  const aipRows = (await db.execute(sql`
    SELECT a.id, a.uuid, a.filename, a.size_bytes, a.node_id
    FROM aips a
    WHERE a.node_id IN ${sql`(${sql.join(treeIds.map((i) => sql`${i}`), sql`, `)})`} AND a.deleted_at IS NULL
  `)) as unknown as Array<Record<string, unknown>>;
  for (const a of aipRows) {
    const attach = gNodes.find((gn) => gn.id === `n${Number(a.node_id)}`);
    if (!attach) continue;
    const x = attach.x + COL_W;
    let y = attach.y;
    while ([...gNodes].some((gn) => gn.x === x && gn.y === y)) y += ROW_H / 1.5;
    gNodes.push({
      id: `a${Number(a.id)}`, kind: 'aip',
      label: { en: String(a.filename ?? String(a.uuid).slice(0, 8)) },
      meta: { sizeBytes: Number(a.size_bytes) }, x, y,
    });
    gEdges.push({ id: `ea${Number(a.id)}`, source: attach.id, target: `a${Number(a.id)}`, kind: 'aip' });
  }

  const relRows = (await db.execute(sql`
    SELECT r.id, r.subject_node_id, rt.name AS type_name, o.id AS oid, o.slug, o.title, o.kind
    FROM relations r
    JOIN terms rt ON rt.id = r.type_term_id
    JOIN nodes o ON o.id = r.object_node_id AND o.deleted_at IS NULL
    WHERE r.subject_node_id IN ${sql`(${sql.join(treeIds.map((i) => sql`${i}`), sql`, `)})`}
  `)) as unknown as Array<Record<string, unknown>>;
  const maxX = Math.max(...gNodes.map((gn) => gn.x)) + COL_W;
  for (const r of relRows) {
    const nid = `n${Number(r.oid)}`;
    if (!gNodes.some((gn) => gn.id === nid)) {
      let y = 0;
      while (gNodes.some((gn) => gn.x === maxX && gn.y === y)) y += ROW_H;
      gNodes.push({ id: nid, kind: 'supporting', label: r.title as I18nText, slug: String(r.slug), x: maxX, y });
    }
    gEdges.push({
      id: `r${Number(r.id)}`, source: `n${Number(r.subject_node_id)}`, target: nid,
      kind: 'relation', label: r.type_name as I18nText,
    });
  }

  return { nodes: gNodes, edges: gEdges };
}
```

```tsx
import { notFound } from 'next/navigation';
import { getLocale, getTranslations } from 'next-intl/server';

          <div id="context-browser-slot" />{/* Task 6: ContextBrowser */}
          <AipList aips={d.aips} locale={locale} t={{
            title: t('aips'), ingested: t('ingested'), empty: t('noAips'),
          }} formatBytes={formatBytes} />
          <RelationList relations={d.relations} locale={locale} t={{
            title: t('relations'), out: t('relOut'), in: t('relIn'),

      </div>
    </div>
  );
}
```

```tsx
'use client';

import { useMemo } from 'react';
import { useRouter } from 'next/navigation';
import { ReactFlow, Background, Controls, type Node, type Edge } from '@xyflow/react';
import '@xyflow/react/dist/style.css';
import type { ContextGraph } from '@binder/db';
import type { Culture } from '@binder/core';
import { pickI18n } from '@binder/core';
import { EDGE_STYLE, NODE_STYLE } from './graph-theme';

export function ContextBrowser({ graph, locale, currentId, title }: {
  graph: ContextGraph; locale: Culture; currentId: number; title: string;
}) {
  const router = useRouter();

  const nodes = useMemo<Node[]>(() => graph.nodes.map((n) => ({
    id: n.id,
    position: { x: n.x, y: n.y },
    data: {
      label: `${n.meta?.componentNumber ? `${n.meta.componentNumber} · ` : ''}${pickI18n(n.label, locale, 'ko')}`,
    },
    style: {
      ...NODE_STYLE[n.kind],
      ...(n.id === `n${currentId}` ? { boxShadow: '0 0 0 3px #93c5fd' } : {}),
    },
    sourcePosition: 'right' as const,
    targetPosition: 'left' as const,
  })), [graph, locale, currentId]);

  const edges = useMemo<Edge[]>(() => graph.edges.map((e) => ({
    id: e.id, source: e.source, target: e.target,
    label: e.label ? pickI18n(e.label, locale, 'ko') : undefined,
    style: { stroke: EDGE_STYLE[e.kind].stroke, strokeDasharray: EDGE_STYLE[e.kind].dash },
    labelStyle: { fontSize: 10 },
  })), [graph, locale]);

  return (
    <section className="rounded-lg border bg-white p-4" data-testid="context-browser">
      <h2 className="mb-3 text-sm font-semibold text-neutral-500">{title}</h2>
      <div style={{ height: 480 }}>
        <ReactFlow
          nodes={nodes}
          edges={edges}
          fitView
          nodesDraggable={false}
          nodesConnectable={false}
          onNodeClick={(_, node) => {
            const g = graph.nodes.find((x) => x.id === node.id);
            if (g?.slug && g.id !== `n${currentId}`) router.push(`/records/${g.slug}`);
          }}
        >
          <Background />
          <Controls showInteractive={false} />
        </ReactFlow>
      </div>
    </section>
  );
}
```
